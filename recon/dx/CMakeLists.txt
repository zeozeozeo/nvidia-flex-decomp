# Cross-build of the reconstructed D3D backend (Windows x64 DLL).
#
#   cmake -S recon/dx -B recon/build-dx -G Ninja \
#         -DCMAKE_TOOLCHAIN_FILE=recon/dx-toolchain-mingw.cmake
#   ninja -C recon/build-dx
#
# The target name and the export list match the shipped binary so the DLL can be
# dropped into a copy of FleX/bin/win64 and exercised with the demo under Wine.
#
# Everything is compiled once into an OBJECT library; the DLL and the
# `d3d_smoke` device test then link the same objects.
cmake_minimum_required(VERSION 3.20)
project(nvflexd3d CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(DX_SOURCES "")
file(GLOB DX_SOURCES CONFIGURE_DEPENDS
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/*.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/contextD3D11/*.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/contextD3D12/*.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/*/*.part*.cpp)

# generated by `flexkit d3dblobs`: the shipped flex.obj shader table
set(DX_GENERATED ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/generated/shaders_flex.cpp)

add_library(nvflexd3d_objects OBJECT ${DX_SOURCES} ${DX_GENERATED})
set_target_properties(nvflexd3d_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(nvflexd3d_objects PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/../include
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/generated
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/contextD3D11
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/contextD3D12
  ${CMAKE_CURRENT_SOURCE_DIR}/../include/nvflex)
target_compile_options(nvflexd3d_objects PRIVATE -O2 -g -ffp-contract=off -fvisibility=default)

add_library(nvflexd3d SHARED $<TARGET_OBJECTS:nvflexd3d_objects>)
set_target_properties(nvflexd3d PROPERTIES OUTPUT_NAME "NvFlexReleaseD3D_x64" PREFIX "")
target_link_libraries(nvflexd3d PRIVATE d3d11 d3d12 dxgi dxguid d3dcompiler user32)
# Link the mingw runtime statically: the shipped tree carries no libgcc/libstdc++
# DLLs, and the drop-in harness only copies FleX's own binaries.
target_link_options(nvflexd3d PRIVATE -static-libgcc -static-libstdc++ -static)

# Export surface: exactly the shipped 83-symbol list from `exports.def`.  Nothing
# else may leak out of the DLL - in particular the GL/CUDA-only entry points that
# the shipped D3D build does not export.  Enforced by `flexkit gate dll`.
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/exports.def)
  target_link_options(nvflexd3d PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/exports.def)
else()
  message(FATAL_ERROR "recon/dx/exports.def is missing")
endif()

# ---------------------------------------------------------------------------
# Device-layer smoke test (Step 3 acceptance)
#
# Drives DeviceD3D11 + ContextD3D11 directly: create a buffer, upload the
# ClearInt parameters, dispatch, read back.  `ninja -C recon/build-dx d3d_smoke`
# then `wine d3d_smoke.exe`; the expected line is `device=ok clear=ok first=7`.
# ---------------------------------------------------------------------------
add_executable(d3d_smoke
  ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/parity/d3d/device_smoke.cpp
  $<TARGET_OBJECTS:nvflexd3d_objects>)
target_link_libraries(d3d_smoke PRIVATE d3d11 d3d12 dxgi dxguid d3dcompiler user32)
target_include_directories(d3d_smoke PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/../include
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/generated
  ${CMAKE_CURRENT_SOURCE_DIR}/../src/dx/contextD3D11)
set_target_properties(d3d_smoke PROPERTIES OUTPUT_NAME "d3d_smoke")
target_link_options(d3d_smoke PRIVATE -static-libgcc -static-libstdc++ -static)
